========================================
  SECRELY — EXACT TERMINAL COMMANDS
========================================

-----------------------------------------
1. PROJECT SETUP (first time only)
-----------------------------------------

# Navigate into the project directory
cd secrely

# Install all dependencies
npm install


-----------------------------------------
2. DEVELOPMENT SERVER
-----------------------------------------

# Start local development server (hot reload)
npm run dev

# App runs at: http://localhost:3000


-----------------------------------------
3. PRODUCTION BUILD
-----------------------------------------

# Create optimised production build
npm run build

# Start the production server (after build)
npm run start


-----------------------------------------
4. LINTING
-----------------------------------------

# Run ESLint
npm run lint


-----------------------------------------
5. DEPENDENCY NOTES
-----------------------------------------

# All dependencies are declared in package.json.
# Running npm install will install everything:
#
# - next@14.2.5
# - react@18, react-dom@18
# - typescript@5
# - tailwindcss@3
# - motion@11 (Framer Motion v11)
# - zustand@4
# - lucide-react@0.383.0
# - clsx, tailwind-merge, class-variance-authority
# - @radix-ui/* (dialog, avatar, tabs, toast, switch, etc.)
# - date-fns@3


-----------------------------------------
6. NO shadcn/ui CLI NEEDED
-----------------------------------------

# This project does NOT use the shadcn/ui CLI.
# All UI components are hand-written in:
#   src/components/ui/
#
# They are built on top of @radix-ui/* primitives
# (which ARE installed via package.json).
# No `npx shadcn-ui init` or `npx shadcn-ui add` required.


-----------------------------------------
7. NO ENVIRONMENT VARIABLES NEEDED
-----------------------------------------

# This MVP requires zero .env configuration.
# All data is mocked locally.
# No API keys, database URLs, or secrets needed.


-----------------------------------------
8. QUICK START (copy-paste)
-----------------------------------------

cd secrely && npm install && npm run dev


-----------------------------------------
9. RESET LOCAL DATA
-----------------------------------------

# To clear all localStorage state (credits, unlocks, auth):
# Open browser DevTools → Application → Local Storage
# Delete all keys starting with "secrely_"
#
# Or run this in browser console:
# Object.keys(localStorage).filter(k => k.startsWith('secrely_')).forEach(k => localStorage.removeItem(k))
# Then refresh the page.
